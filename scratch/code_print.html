1144 for (let j = 0; j <= a.length; j++) matrix[0][j] = j;
1145 for (let i = 1; i <= b.length; i++) {
1146 for (let j = 1; j <= a.length; j++) {
1147 if (b.charAt(i - 1) === a.charAt(j - 1)) {
1148 matrix[i][j] = matrix[i - 1][j - 1];

1150 matrix[i][j] = Math.min(matrix[i - 1][j - 1] + 1, matrix[i][j - 1] + 1, matrix[i - 1][j] + 1);

2849 let heldoutTotalExpected = 0, heldoutTotalExtracted = 0, heldoutTruePositives = 0;

2855 const expectedMeds = testCase.truth.medications || [];
2856

2858 expectedMeds.forEach(exp => {
2859 const expName = exp.name.toLowerCase();

2861 tp++;
2862 }
2863 if (exp.dose) {
2864 totalDosesEvaluated++;
2865 const foundMed = parsed.medications.find(m => m.name.toLowerCase() === expName);
2866 if (foundMed && foundMed.dose.replace(/\s+/g, '').toLowerCase() === exp.dose.replace(/\s+/g, '').toLowerCase()) {
2867 doseExactMatchCount++;
2868 }
2869 }
2870 });
2871 
2872 if (testCase.type === 'tuning') {
2873 tuningTotalExpected += expectedMeds.length;
2874 tuningTotalExtracted += extractedNames.length;
2875 tuningTruePositives += tp;
2876 } else {
2877 heldoutTotalExpected += expectedMeds.length;
2878 heldoutTotalExtracted += extractedNames.length;
2879 heldoutTruePositives += tp;

2884 const tuningRecall = (tuningTruePositives / (tuningTotalExpected || 1)) * 100;
